## Tools
kite.cmd={runtime.tools.bundle.path}/kite
sdcc.cmd=sdcc
ar.cmd=sdar
objcopy.cmd=sdobjcopy
upload.cmd={runtime.tools.bundle.path}/nvtispflash
upload.cmd.windows={runtime.tools.bundle.path}/NuvoISP.exe

## Flags
compiler.flags=-mmcs51 --model-small --iram-size 256 --xram-size 768 --code-size 18432

## Preprocess (for ctags)
recipe.preproc.macros={kite.cmd} pp {includes} "{source_file}" "{preprocessed_file_path}"

## For compiling core
recipe.c.o.pattern={sdcc.cmd} {compiler.flags} {includes} -c "{source_file}" -o "{object_file}"

## For compiling sketch
recipe.cpp.o.pattern={sdcc.cmd} {compiler.flags} {includes} -x c -c "{source_file}" -o "{object_file}"

## Build core archive
recipe.ar.pattern={ar.cmd} -rcs "{archive_file_path}" "{object_file}"

## Link (object_files is sketch + libraries)
ihexfile="{build.path}/{build.project_name}.ihx"
recipe.c.combine.pattern={kite.cmd} ld {object_files} "{archive_file_path}" {ihexfile}

## Convert ihx → bin
objcopy.flags=-I ihex -O binary
apromfile="{build.path}/{build.project_name}.bin"
recipe.objcopy.bin.pattern={objcopy.cmd} {objcopy.flags} {ihexfile} {apromfile}

## Upload
upload.pattern={upload.cmd} --serial-device "{serial.port}" --aprom-file {apromfile}
upload.pattern.windows={upload.cmd} -interface UART "{serial.port}" -aprom "{build.path}/{build.project_name}.bin" -erase -run
